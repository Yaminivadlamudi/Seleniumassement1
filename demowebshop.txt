2.Identify the webelement(14.1-inch laptop)-->identify add to cart button capture the size and them click on the buuton.
3.Click on shopping cart link.
4.verify remove check box is selected or not and capture the complete information using getRect().
5.Verify Alert message is displayed or not before and after clicking on Apply coupon button

package demowebshop;

import java.io.File;
import java.io.IOException;
import java.time.Duration;

import org.openqa.selenium.By;
import org.openqa.selenium.OutputType;
import org.openqa.selenium.Rectangle;
import org.openqa.selenium.TakesScreenshot;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.io.FileHandler;

public class Testcase3 {

	public static void main(String[] args) throws InterruptedException, IOException {

		WebDriver driver = new ChromeDriver();
		driver.manage().window().maximize();
		driver.manage().timeouts().implicitlyWait(Duration.ofSeconds(10));

		driver.get("https://demowebshop.tricentis.com/");

		WebElement laptop = driver.findElement(By.xpath("//a[text()='14.1-inch Laptop']"));

		WebElement add = driver.findElement(By.xpath("//a[text()='14.1-inch Laptop']/ancestor::div[@class='product-item']//input[@value='Add to cart']"));

		System.out.println("Add to cart width = " + add.getSize().getWidth());
		System.out.println("Add to cart height = " + add.getSize().getHeight());

		add.click();

		Thread.sleep(2000);

		driver.findElement(By.xpath("//span[text()='Shopping cart']")).click();

		Thread.sleep(2000);

		WebElement remove = driver.findElement(By.name("removefromcart"));

		if(remove.isSelected())
		{
			System.out.println("Remove checkbox is selected");
		}
		else
		{
			System.out.println("Remove checkbox is not selected");
		}

		Rectangle r = remove.getRect();

		System.out.println("X = " + r.getX());
		System.out.println("Y = " + r.getY());
		System.out.println("Width = " + r.getWidth());
		System.out.println("Height = " + r.getHeight());

		if(driver.findElements(By.className("message-error")).size() > 0)
		{
			System.out.println("Message is displayed before applying coupon");
		}
		else
		{
			System.out.println("Message is not displayed before applying coupon");
		}

		WebElement coupon = driver.findElement(By.id("discountcouponcode"));
		coupon.sendKeys("TEST123");

		driver.findElement(By.name("applydiscountcouponcode")).click();

		Thread.sleep(2000);

		if(driver.findElements(By.className("message-error")).size() > 0)
		{
			System.out.println("Message is displayed after applying coupon");
		}
		else
		{
			System.out.println("Message is not displayed after applying coupon");
		}

		WebElement image = driver.findElement(By.xpath("//img[contains(@alt,'14.1-inch Laptop')]"));

		File src = image.getScreenshotAs(OutputType.FILE);
		File dest = new File("./laptop.png");

		FileHandler.copy(src, dest);

		System.out.println("Laptop screenshot taken");

		driver.quit();
	}
}
